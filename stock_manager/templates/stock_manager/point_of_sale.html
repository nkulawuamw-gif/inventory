function renderCart() { const container = document.getElementById('cartItems'); const subtotalEl = document.getElementById('subtotalDisplay'); const totalEl = document.getElementById('totalDisplay'); const btn = document.getElementById('completeBtn'); const count = document.getElementById('cartCount'); count.textContent = cart.reduce((s, x) => s + x.qty, 0); if (cart.length === 0) { container.innerHTML = `
No items added yet
`; subtotalEl.textContent = 'MWK 0.00'; totalEl.textContent = 'MWK 0.00'; btn.disabled = true; btn.innerHTML = ' Add items to start'; document.getElementById('paymentSection').style.display = 'none'; calcChange(); return; } let html = ''; let subtotal = 0; cart.forEach(x => { const lineTotal = x.price * x.qty; subtotal += lineTotal; html += `
${x.name}
MWK ${x.price.toFixed(2)}
${x.qty}
MWK ${lineTotal.toFixed(2)}
`; }); container.innerHTML = html; subtotalEl.textContent = `MWK ${subtotal.toFixed(2)}`; totalEl.textContent = `MWK ${subtotal.toFixed(2)}`; btn.disabled = false; btn.innerHTML = ' Complete Sale'; document.getElementById('paymentSection').style.display = 'block'; calcChange(); }
